import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

# ==============================================================================
# PROGRAM 1: Basic Perceptron & Binary Classification
# Objective: Build a basic perceptron/single-layer neural network to understand 
# fundamental mechanics and decision boundaries for binary classification.
# ==============================================================================
print("--- Running Program 1 ---")
# 1. Dummy Data (Logical OR Gate)
X_p1 = np.array([[0,0], [0,1], [1,0], [1,1]], dtype=float)
y_p1 = np.array([[0], [1], [1], [1]], dtype=float)

# 2. Build and Train
model_p1 = tf.keras.Sequential([
    tf.keras.layers.Dense(1, input_shape=(2,), activation='sigmoid')
])
model_p1.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
model_p1.fit(X_p1, y_p1, epochs=500, verbose=0)

# 3. Evaluate
loss_1, acc_1 = model_p1.evaluate(X_p1, y_p1, verbose=0)
print(f"Program 1 - Accuracy: {acc_1:.4f}\n")


# ==============================================================================
# PROGRAM 2: Deep Neural Network for Handwriting Recognition
# Objective: Build a deep neural network to read handwriting using the MNIST 
# dataset. Compile, train, evaluate, and visualize the training history.
# ==============================================================================
print("--- Running Program 2 ---")
# 1. Load and Prep Data
(X_train_p2, y_train_p2), (X_test_p2, y_test_p2) = tf.keras.datasets.mnist.load_data()
X_train_p2 = X_train_p2.reshape(-1, 784) / 255.0
X_test_p2 = X_test_p2.reshape(-1, 784) / 255.0

# 2. Build Model
model_p2 = tf.keras.Sequential([
    tf.keras.layers.Dense(128, activation='relu', input_shape=(784,)),
    tf.keras.layers.Dense(64, activation='relu'),
    tf.keras.layers.Dense(10, activation='softmax')
])

# 3. Compile and Train
model_p2.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
history_p2 = model_p2.fit(X_train_p2, y_train_p2, epochs=10, batch_size=32, validation_split=0.2, verbose=0)

# 4. Evaluate and Plot
loss_p2, acc_p2 = model_p2.evaluate(X_test_p2, y_test_p2, verbose=0)
print(f"Program 2 - Test Accuracy: {acc_p2:.4f}")

plt.figure()
plt.plot(history_p2.history['accuracy'], label='Train Accuracy')
plt.plot(history_p2.history['val_accuracy'], label='Validation Accuracy')
plt.xlabel('Epochs')
plt.ylabel('Accuracy')
plt.title('Program 2: Model Accuracy')
plt.legend()
plt.show()


# ==============================================================================
# PROGRAM 3: Activation Functions Showdown
# Objective: Explore how different activation functions (Sigmoid, ReLU, and Tanh) 
# affect a neural network's ability to learn using the MNIST dataset.
# ==============================================================================
print("--- Running Program 3 ---")
def create_model_p3(af):
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(128, activation=af, input_shape=(784,)),
        tf.keras.layers.Dense(64, activation=af),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
    return model

activation_functions = ['sigmoid', 'tanh', 'relu']
results_p3 = {}

for activation in activation_functions:
    print(f"Training with {activation}...")
    model = create_model_p3(activation)
    history = model.fit(X_train_p2, y_train_p2, epochs=5, batch_size=32, validation_data=(X_test_p2, y_test_p2), verbose=0)
    results_p3[activation] = history

plt.figure(figsize=(10, 6))
for activation in activation_functions:
    plt.plot(results_p3[activation].history['val_accuracy'], label=f"{activation} Validation")
plt.title('Program 3: Activation Function Showdown')
plt.xlabel('Epochs')
plt.ylabel('Validation Accuracy')
plt.legend()
plt.show()


# ==============================================================================
# PROGRAM 4: The Optimizers (SGD vs. RMSprop vs. Adam)
# Objective: Build a CNN to classify color images from the CIFAR-10 dataset, 
# and compare three different optimizers (SGD, Adam, RMSprop).
# ==============================================================================
print("--- Running Program 4 ---")
(X_train_p4, y_train_p4), (X_test_p4, y_test_p4) = tf.keras.datasets.cifar10.load_data()
X_train_p4, X_test_p4 = X_train_p4 / 255.0, X_test_p4 / 255.0

def build_and_train_p4(optimizer_name):
    model = tf.keras.Sequential([
        tf.keras.layers.Conv2D(32, (3,3), activation='relu', input_shape=(32, 32, 3)),
        tf.keras.layers.MaxPooling2D((2,2)),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(128, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    model.compile(optimizer=optimizer_name, loss='sparse_categorical_crossentropy', metrics=['accuracy'])
    history = model.fit(X_train_p4, y_train_p4, epochs=5, batch_size=64, validation_data=(X_test_p4, y_test_p4), verbose=0)
    return history

optimizers = ['sgd', 'rmsprop', 'adam']
histories_p4 = {}

for opt in optimizers:
    print(f"Training with {opt}...")
    histories_p4[opt] = build_and_train_p4(opt)

plt.figure(figsize=(10, 6))
for name in optimizers:
    plt.plot(histories_p4[name].history['val_accuracy'], label=f'{name.upper()} Val', linestyle='--')
plt.title('Program 4: Optimizer Race')
plt.xlabel('Epoch')
plt.ylabel('Validation Accuracy')
plt.legend()
plt.show()


# ==============================================================================
# PROGRAM 5: Regularization Lifeguards (L2 & Dropout)
# Objective: Apply L2 Regularization and Dropout to a neural network to prevent 
# overfitting on the Fashion MNIST dataset.
# ==============================================================================
print("--- Running Program 5 ---")
(X_train_p5, y_train_p5), (X_test_p5, y_test_p5) = tf.keras.datasets.fashion_mnist.load_data()
X_train_p5 = X_train_p5.reshape(-1, 784) / 255.0
X_test_p5 = X_test_p5.reshape(-1, 784) / 255.0

model_p5 = tf.keras.Sequential([
    tf.keras.layers.Dense(256, activation='relu', kernel_regularizer=tf.keras.regularizers.l2(1e-4), input_shape=(784,)),
    tf.keras.layers.Dropout(0.5),
    tf.keras.layers.Dense(128, activation='relu', kernel_regularizer=tf.keras.regularizers.l2(1e-4)),
    tf.keras.layers.Dropout(0.5),
    tf.keras.layers.Dense(10, activation='softmax')
])

model_p5.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
history_p5 = model_p5.fit(X_train_p5, y_train_p5, epochs=10, validation_data=(X_test_p5, y_test_p5), verbose=0)

plt.figure()
plt.plot(history_p5.history["loss"], label="Train Loss")
plt.plot(history_p5.history["val_loss"], label="Validation Loss")
plt.title("Program 5: Effect of Regularization on Loss")
plt.xlabel("Epochs")
plt.ylabel("Loss")
plt.legend()
plt.show()


# ==============================================================================
# PROGRAM 6: Convolutional Neural Network (CNN) for Image Recognition
# Objective: Build a CNN for Image Recognition on MNIST and visualize the 
# learned filters (feature maps).
# ==============================================================================
print("--- Running Program 6 ---")
(X_train_p6, y_train_p6), (X_test_p6, y_test_p6) = tf.keras.datasets.mnist.load_data()
X_train_p6 = X_train_p6.reshape(-1, 28, 28, 1) / 255.0
X_test_p6 = X_test_p6.reshape(-1, 28, 28, 1) / 255.0

inputs = tf.keras.Input(shape=(28, 28, 1))
x = tf.keras.layers.Conv2D(32, (3,3), activation='relu')(inputs)
x = tf.keras.layers.MaxPooling2D((2,2))(x)
x = tf.keras.layers.Conv2D(64, (3,3), activation='relu')(x)
x = tf.keras.layers.MaxPooling2D((2,2))(x)
x = tf.keras.layers.Flatten()(x)
x = tf.keras.layers.Dense(64, activation='relu')(x)
outputs = tf.keras.layers.Dense(10, activation='softmax')(x)

model_p6 = tf.keras.Model(inputs=inputs, outputs=outputs)
model_p6.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
model_p6.fit(X_train_p6, y_train_p6, epochs=3, batch_size=64, validation_split=0.2, verbose=0)

# Visualization
layer_outputs = [layer.output for layer in model_p6.layers if isinstance(layer, tf.keras.layers.Conv2D)]
activation_model = tf.keras.models.Model(inputs=model_p6.input, outputs=layer_outputs)
img_p6 = X_test_p6[0].reshape(1, 28, 28, 1)
activations = activation_model.predict(img_p6)

fig, axes = plt.subplots(1, 6, figsize=(12, 6))
for i in range(6):
    axes[i].imshow(activations[0][0, :, :, i], cmap='viridis')
    axes[i].axis('off')
plt.suptitle('Program 6: CNN Filter Visualizations')
plt.show()


# ==============================================================================
# PROGRAM 7: Recurrent Neural Networks (RNNs) for Text
# Objective: Build an RNN to perform sentiment analysis (classifying text 
# as positive or negative) using the IMDB dataset.
# ==============================================================================
print("--- Running Program 7 ---")
from tensorflow.keras.preprocessing.sequence import pad_sequences

(X_train_p7, y_train_p7), (X_test_p7, y_test_p7) = tf.keras.datasets.imdb.load_data(num_words=10000)
X_train_p7 = pad_sequences(X_train_p7, maxlen=100)
X_test_p7 = pad_sequences(X_test_p7, maxlen=100)

model_p7 = tf.keras.Sequential([
    tf.keras.layers.Embedding(input_dim=10000, output_dim=128, input_length=100),
    tf.keras.layers.SimpleRNN(units=64, activation='relu'),
    tf.keras.layers.Dense(units=1, activation='sigmoid')
])

model_p7.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
model_p7.fit(X_train_p7, y_train_p7, epochs=3, batch_size=64, validation_split=0.2, verbose=0)

loss_p7, acc_p7 = model_p7.evaluate(X_test_p7, y_test_p7, verbose=0)
print(f"Program 7 - Final Test Accuracy: {acc_p7:.4f}\n")


# ==============================================================================
# PROGRAM 8: Transfer Learning (VGG16)
# Objective: Understand how to use pre-trained models for transfer learning.
# Load a pre-trained model (VGG16). Fine-tune the model on a custom dataset 
# (adapted to CIFAR-10). Evaluate performance and visualize predictions.
# ==============================================================================
print("--- Running Program 8 ---")
(X_train_p8, y_train_p8), (X_test_p8, y_test_p8) = tf.keras.datasets.cifar10.load_data()
X_train_p8, X_test_p8 = X_train_p8 / 255.0, X_test_p8 / 255.0
y_train_p8 = tf.keras.utils.to_categorical(y_train_p8, 10)
y_test_p8 = tf.keras.utils.to_categorical(y_test_p8, 10)

model_p8 = tf.keras.Sequential([
    tf.keras.applications.VGG16(include_top=False, weights='imagenet', input_shape=(32, 32, 3)),
    tf.keras.layers.Flatten(),
    tf.keras.layers.Dense(512, activation='relu'),
    tf.keras.layers.Dense(10, activation='softmax')
])

# Freeze the VGG16 base to save time
model_p8.layers[0].trainable = False

model_p8.compile(optimizer='sgd', loss='categorical_crossentropy', metrics=['accuracy'])
model_p8.fit(X_train_p8, y_train_p8, epochs=1, validation_split=0.2, verbose=0)

loss_p8, acc_p8 = model_p8.evaluate(X_test_p8, y_test_p8, verbose=0)
print(f"Program 8 - Final Test Accuracy: {acc_p8:.4f}")

cifar_labels = ['Airplane', 'Automobile', 'Bird', 'Cat', 'Deer', 
                'Dog', 'Frog', 'Horse', 'Ship', 'Truck']

num_images = 5
random_indices = np.random.choice(len(X_test_p8), num_images, replace=False)
selected_images = X_test_p8[random_indices]
true_labels = y_test_p8[random_indices]

predictions = model_p8.predict(selected_images, verbose=0)
predicted_classes = np.argmax(predictions, axis=1)

plt.figure(figsize=(15, 5))
for i in range(num_images):
    plt.subplot(1, num_images, i + 1)
    plt.imshow(selected_images[i])
    plt.axis('off')
    
    actual = cifar_labels[np.argmax(true_labels[i])]
    predicted = cifar_labels[predicted_classes[i]]
    color = 'green' if actual == predicted else 'red'
    plt.title(f"Actual: {actual}\nPred: {predicted}", fontsize=12, color=color)

plt.suptitle('Program 8: VGG16 Transfer Learning Predictions')
plt.tight_layout()
plt.show()
